+ import { Readable } from "stream";
+ 
+ export const DEFAULT_CHUNK_SIZE = 5 * 1024 * 1024;
+ 
+ export interface Chunk {
+   index: number;
+   start: number;
+   end: number; // exclusive
+   etag?: string;
+   uploaded?: boolean;
+ }
+ 
+ export function planChunks(byteLength: number, chunkSize = DEFAULT_CHUNK_SIZE): Chunk[] {
+   if (byteLength <= 0) return [];
+   const out: Chunk[] = [];
+   let i = 0, offset = 0;
+   while (offset < byteLength) {
+     const end = Math.min(byteLength, offset + chunkSize);
+     out.push({ index: i, start: offset, end });
+     offset = end; i++;
+   }
+   return out;
+ }
+ 
+ export function mergeProgress(plan: Chunk[], known: Chunk[]): Chunk[] {
+   const map = new Map(known.map(k => [k.index, k]));
+   return plan.map(p => ({ ...p, ...(map.get(p.index) ?? {}) }));
+ }
+ 
+ export function nextChunk(plan: Chunk[]): Chunk | null {
+   for (const c of plan) if (!c.uploaded) return c;
+   return null;
+ }
+ 
+ export async function* sliceReadable(src: Readable, plan: Chunk[]) {
+   const buf: Buffer[] = [];
+   let bytesQueued = 0;
+   let cursor = 0;
+   for await (const piece of src) {
+     buf.push(piece);
+     bytesQueued += piece.length;
+     while (cursor < plan.length && bytesQueued >= plan[cursor].end) {
+       const need = plan[cursor].end - plan[cursor].start;
+       const joined = Buffer.concat(buf, bytesQueued);
+       yield { chunk: plan[cursor], body: joined.subarray(plan[cursor].start, plan[cursor].end) };
+       cursor++;
+     }
+   }
